# Requires: fitted ALR at outputs/Emulator/ALR and 20CR PCs.
import importlib
import os
import numpy as np
import pandas as pd
import xarray as xr
import utils.bmu_bootstrap_timeseries as bmu_bt
importlib.reload(bmu_bt)

from bluemath_tk.teslakit.alr import ALR_WRP
import utils.alr_plotting as alr_plotting
importlib.reload(alr_plotting)

POINT = (36.200, -75.714)
OUT_DIR = "outputs/test"
SOURCE_DIR = "../02_Wind_Metamodel/outputs/merged_500m_binwaves_bmus"  # same product as CenturyHindcast
CENTURY_DIR = "outputs/century"
VARS = ["hs", "tp", "dp"]
os.makedirs(OUT_DIR, exist_ok=True)

# Clusters for historical BMU pools
waveClusters_and_pcs = pd.read_csv(
    "outputs/PCA/pcs_clusters_ordered_ERA5.csv", index_col=0, parse_dates=False
)
waveClusters_and_pcs.index = pd.to_datetime(waveClusters_and_pcs.index)

# ALR simulate over 20CR PC period (same workflow as cells above)
alr_model = ALR_WRP("outputs/Emulator/ALR")
alr_model.LoadModel()
alr_model.LoadBmus_Fit()

pcs = pd.read_csv("outputs/PCA/20cr_era5_pcs.csv", index_col=0, parse_dates=True)
pcs.index = pd.to_datetime(pcs.index).floor("D")
covar_sim = xr.Dataset(
    data_vars=dict(
        cov_values=(("n_sim", "time", "n_covariates"), pcs.values[None, ...]),
        cov_names=(("n_sim", "n_covariates"), np.array([pcs.columns.values])),
    ),
    coords=dict(
        n_sim=np.arange(1),
        time=pcs.index,
        n_covariates=np.arange(pcs.shape[1]),
    ),
)

simulated_daily_bmus = alr_model.Simulate(
    num_sims=1,
    time_sim=covar_sim.time.values,
    xds_covars_sim=covar_sim.isel(n_sim=0),
)
simulated_daily_bmus.to_netcdf(os.path.join(OUT_DIR, "simulated_daily_bmus_test.nc"))
print("Saved", os.path.join(OUT_DIR, "simulated_daily_bmus_test.nc"))

selection = create_bmu_bootstrap_waves_winds_for_merged_sites(
    hindcast_waves_winds_dir=SOURCE_DIR,
    simulated_daily_bmus=simulated_daily_bmus,
    waveclusters_and_pcs=waveClusters_and_pcs,
    output_dir=OUT_DIR,
    monthly_conditioning=True,
    skip_existing_outputs=False,
    sim_indices=0,
    n_workers=3,
    selected_coordinate=POINT,
    selected_variables=VARS,
)
print(selection)
print("Wrote:", sorted(f for f in os.listdir(OUT_DIR) if f.endswith(".nc")))
